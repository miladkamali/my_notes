#+begin_src gnuplot :results file raw :file ./temp.png :post attr_wrap(data=*this*, width=500)
  reset
  set title "Test plot"
  do for [i=0:20] {
      
      # Calculate a "shift" value (phase) for the sine wave
      shift = i * 0.1 * pi
      
      # 3. Create a dynamic filename (e.g., frame_000.png, frame_001.png)
      # %03d pads the number with zeros (001 instead of 1), which helps sorting later.
      outfile = sprintf('frame_%03d.png', i)
      set output outfile
      
      # 4. Plot the current frame
      set title sprintf("Phase shift: %.2f pi", i * 0.1)
      plot sin(x + shift) lw 3 title "Moving Sine Wave"
      
      # 5. Close the file (optional but good practice)
      unset output
  }
#+end_src
