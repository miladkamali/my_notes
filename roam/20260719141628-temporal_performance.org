 #+begin_src shell
   ./temporal workflow list   --query 'WorkflowType="GreetingWorkflow" and ExecutionStatus="Completed"'   --limit 1000   --output json |   jq -r '.[] | (.executionDuration // .execution_duration) | rtrimstr("s") | tonumber' |   python3 -c '
import sys, statistics, math
data = sorted([float(x.strip()) for x in sys.stdin if x.strip()])
if not data:
    print("No completed workflows found."); sys.exit(0)
n = len(data)
def p(pct): return data[max(0, math.ceil(pct / 100.0 * n) - 1)]
print(f"Count : {n}")
print(f"Avg   : {statistics.mean(data):.4f}s")
print(f"Median: {statistics.median(data):.4f}s")
print(f"P95   : {p(95):.4f}s")
print(f"P99   : {p(99):.4f}s")
'

        // If activities are I/O bound (network calls, DB queries), set this high (e.g., 200-1000).
        // If CPU bound, keep it closer to your logical CPU core count.
        .activity_slot_supplier(Arc::new(FixedSizeSlotSupplier::new(500)))

        // OPTIMIZATION 2: Workflow Concurrency
        // Workflows are highly efficient. You can usually leave this at 100 or slightly increase it.
        // Note: For caching, the worker must be willing to hand out at least two slots.
        .workflow_slot_supplier(Arc::new(FixedSizeSlotSupplier::new(500)))
