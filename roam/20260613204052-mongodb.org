** backup and restore

Full database
mongodump \
  --uri="mongodb://USER:PASSWORD@HOST:27017/DATABASE" \
  --out=/tmp/mongo-backup-$(date +%F)
Single database
mongodump \
  --uri="mongodb://USER:PASSWORD@HOST:27017/mydb" \
  --db=mydb \
  --out=/tmp/mongo-backup
Single collection
mongodump \
  --uri="mongodb://USER:PASSWORD@HOST:27017/mydb" \
  --db=mydb \
  --collection=user_profile \
  --out=/tmp/mongo-backup

  tar -czf mongo-backup-$(date +%F).tar.gz -C /tmp mongo-backup-$(date +%F)


  mongodump --uri="mongodb://localhost:27017" --gzip --archive=mongo-$(date +%F).gz

  mongorestore --uri="mongodb://localhost:27017" --gzip --archive=mongo-2026-06-13.gz
