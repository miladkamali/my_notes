#+begin_src go
  import (
  	"errors"
  	"fmt"
  )

  func main(){
  	printMe()
  	var result,remainder,err = intDivision(30,0)
  	switch{
  		case err!=nil:
  		    fmt.Println(err.Error())
  		case remainder==0:
  		    fmt.Printf("the result of integer division is %v",result)
  		default:
  		    fmt.Printf("the result is %v with remainder %v",result,remainder)
  	}
  }
  func printMe(){
  	fmt.Println("hello World")
  }

  func intDivision(numerator int, denominator int) (int, int, error) {
  	var err error
  	if denominator == 0 {
  		err = errors.New("can not divide by zero")
  		return 0, 0, err
  	}
  	var result int = numerator / denominator
  	var remainder int = numerator % denominator
  	return result, remainder, err
  }
#+end_src
